import { initializeApp } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-app.js"; import { getFirestore, collection, getDocs } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-firestore.js"; // আপনার admin.js এ থাকা Firebase Config টি এখানে দেওয়া হলো const firebaseConfig = { apiKey: "YOUR_PROJECT_ID", // আপনার আসল apiKey যদি থাকে তবে তা বসাবেন authDomain: "jobnest-india.firebaseapp.com", projectId: "jobnest-india", storageBucket: "jobnest-india.appspot.com", messagingSenderId: "100000000000", appId: "1:100000000000:web:123456" }; const app = initializeApp(firebaseConfig); const db = getFirestore(app); async function fetchAdmitCards() { const container = document.getElementById("admitCardContainer"); if (!container) return; try { const querySnapshot = await getDocs(collection(db, "jobs")); container.innerHTML = ""; let hasAdmitCards = false; querySnapshot.forEach((doc) => { const data = doc.data(); // শুধুমাত্র Admit Card ক্যাটাগরির পোস্ট অথবা যেসব পোস্টে category নির্ধারণ করা নেই সেগুলো দেখাবে if (!data.category || data.category === "admit-card") { hasAdmitCards = true; const card = document.createElement("div"); card.className = "job-card"; card.innerHTML = ` 🟢 NEW

${data.title || 'Admit Card'}

Department: ${data.company || 'N/A'}

Details / Exam Date: ${data.salary || 'N/A'}

Location: ${data.location || 'N/A'}

Download Admit Card
`; container.appendChild(card); } }); if (!hasAdmitCards) { container.innerHTML = "

No Admit Cards found.

"; } } catch (error) { console.error("Error fetching admit cards:", error); container.innerHTML = "

Error loading admit cards.

"; } } document.addEventListener("DOMContentLoaded", fetchAdmitCards);
